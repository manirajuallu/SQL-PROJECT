# GYM MANAGEMENT SYSTEM – MYSQL PROJECT

-- =====================================================
-- PART 1: DATABASE AND TABLES CREATION
-- =====================================================

-- Delete the existing database if it already exists.
-- WARNING: This deletes all existing data in this database.
DROP DATABASE IF EXISTS gym_management_system;

-- Create a new database for the Gym Management System.
CREATE DATABASE gym_management_system;

-- Select the database to use.
USE gym_management_system;


-- -----------------------------------------------------
-- TABLE 1: MEMBERS
-- Purpose: Store gym members' personal information.
-- -----------------------------------------------------

CREATE TABLE members (
    member_id INT AUTO_INCREMENT PRIMARY KEY, -- Unique ID for each member
    member_name VARCHAR(100),                 -- Name of the member
    gender VARCHAR(10),                       -- Gender of the member
    phone VARCHAR(15),                        -- Contact number
    city VARCHAR(50),                         -- City where the member lives
    join_date DATE                            -- Date the member joined the gym
);


-- -----------------------------------------------------
-- TABLE 2: TRAINERS
-- Purpose: Store gym trainers' information.
-- -----------------------------------------------------

CREATE TABLE trainers (
    trainer_id INT AUTO_INCREMENT PRIMARY KEY, -- Unique ID for each trainer
    trainer_name VARCHAR(100),                 -- Name of the trainer
    specialization VARCHAR(100),               -- Trainer's area of expertise
    phone VARCHAR(15)                          -- Trainer's contact number
);


-- -----------------------------------------------------
-- TABLE 3: MEMBERSHIP PLANS
-- Purpose: Store available gym membership plans and fees.
-- -----------------------------------------------------

CREATE TABLE membership_plans (
    plan_id INT AUTO_INCREMENT PRIMARY KEY, -- Unique ID for each plan
    plan_name VARCHAR(50),                  -- Name of the plan
    duration_months INT,                    -- Duration of the plan in months
    fee DECIMAL(10,2)                       -- Membership fee
);


-- -----------------------------------------------------
-- TABLE 4: MEMBERSHIPS
-- Purpose: Connect members, plans, and trainers.
-- -----------------------------------------------------

CREATE TABLE memberships (
    membership_id INT AUTO_INCREMENT PRIMARY KEY, -- Unique membership ID
    member_id INT,                                -- ID of the member
    plan_id INT,                                  -- ID of the selected plan
    trainer_id INT,                               -- ID of the assigned trainer
    start_date DATE,                              -- Membership starting date
    end_date DATE,                                -- Membership ending date

    -- Connect the membership to an existing member.
    FOREIGN KEY (member_id) REFERENCES members(member_id),

    -- Connect the membership to an existing membership plan.
    FOREIGN KEY (plan_id) REFERENCES membership_plans(plan_id),

    -- Connect the membership to an existing trainer.
    FOREIGN KEY (trainer_id) REFERENCES trainers(trainer_id)
);


-- -----------------------------------------------------
-- TABLE 5: PAYMENTS
-- Purpose: Store payments made by gym members.
-- -----------------------------------------------------

CREATE TABLE payments (
    payment_id INT AUTO_INCREMENT PRIMARY KEY, -- Unique payment ID
    member_id INT,                             -- ID of the member making payment
    amount DECIMAL(10,2),                      -- Amount paid by the member
    payment_date DATE,                         -- Date of payment

    -- Allow only Cash, UPI, or Card as payment methods.
    payment_mode ENUM('Cash','UPI','Card'),

    -- Connect the payment to an existing member.
    FOREIGN KEY (member_id) REFERENCES members(member_id)
);


-- -----------------------------------------------------
-- TABLE 6: ATTENDANCE
-- Purpose: Record members' daily attendance.
-- -----------------------------------------------------

CREATE TABLE attendance (
    attendance_id INT AUTO_INCREMENT PRIMARY KEY, -- Unique attendance ID
    member_id INT,                                -- ID of the member
    attendance_date DATE,                         -- Date of attendance

    -- Allow only Present or Absent as attendance status.
    status ENUM('Present','Absent'),

    -- Connect attendance records to an existing member.
    FOREIGN KEY (member_id) REFERENCES members(member_id)
);

-- PART 3: CREATING INDEXES
-- =====================================================

-- Index to help search members by city.
CREATE INDEX idx_member_city
ON members(city);

-- Index to help search payments by payment date.
CREATE INDEX idx_payment_date
ON payments(payment_date);

-- Index to help search memberships by member ID.
CREATE INDEX idx_membership_member
ON memberships(member_id);

-- Index to help search memberships by trainer ID.
CREATE INDEX idx_membership_trainer
ON memberships(trainer_id);

-- Index to help search attendance records by member ID.
CREATE INDEX idx_attendance_member
ON attendance(member_id);


-- =====================================================
-- PART 4: CREATING VIEWS
-- =====================================================

-- VIEW 1: MEMBER DETAILS
-- Purpose: Display member, plan, trainer, and membership dates
-- together in one result.

CREATE VIEW view_member_details AS
SELECT
    m.member_id,
    m.member_name,
    mp.plan_name,
    t.trainer_name,
    ms.start_date,
    ms.end_date
FROM memberships ms

-- Join memberships with the members table.
JOIN members m
    ON ms.member_id = m.member_id

-- Join memberships with the membership plans table.
JOIN membership_plans mp
    ON ms.plan_id = mp.plan_id

-- Join memberships with the trainers table.
JOIN trainers t
    ON ms.trainer_id = t.trainer_id;


-- VIEW 2: TOTAL REVENUE
-- Purpose: Calculate the total amount collected from payments.

CREATE VIEW view_total_revenue AS
SELECT
    SUM(amount) AS total_revenue
FROM payments;


-- VIEW 3: TRAINER-WISE MEMBER COUNT
-- Purpose: Display the number of members assigned to each trainer.
-- LEFT JOIN also includes trainers who have no memberships.

CREATE VIEW view_trainer_members AS
SELECT
    t.trainer_name,
    COUNT(ms.member_id) AS total_members
FROM trainers t
LEFT JOIN memberships ms
    ON t.trainer_id = ms.trainer_id
GROUP BY t.trainer_id, t.trainer_name;


-- =====================================================
-- PART 5: CREATING STORED PROCEDURES
-- =====================================================

-- PROCEDURE 1: MEMBER MEMBERSHIP DETAILS
-- Purpose: Display membership details for a particular member.
-- IN parameter p_member_id receives the member's ID.

DELIMITER $$

CREATE PROCEDURE sp_member_details(
    IN p_member_id INT
)
BEGIN
    -- Retrieve the member's details from the view.
    SELECT *
    FROM view_member_details
    WHERE member_id = p_member_id;
END $$

DELIMITER ;


-- PROCEDURE 2: TOTAL REVENUE REPORT
-- Purpose: Calculate the total revenue from all payments.

DELIMITER $$

CREATE PROCEDURE sp_total_revenue()
BEGIN
    -- Add all payment amounts and display the total.
    SELECT
        SUM(amount) AS revenue
    FROM payments;
END $$

DELIMITER ;


-- PROCEDURE 3: ADD PAYMENT
-- Purpose: Add a new payment for a member.
-- The payment date is automatically set to today's date.
-- The trigger created in Part 6 validates the payment amount.

DELIMITER $$

CREATE PROCEDURE sp_add_payment(
    IN p_member_id INT,       -- Member making the payment
    IN p_amount DECIMAL(10,2), -- Amount to be paid
    IN p_mode VARCHAR(10)      -- Payment method
)
BEGIN
    -- Insert the new payment into the payments table.
    INSERT INTO payments
        (member_id, amount, payment_date, payment_mode)
    VALUES
        (p_member_id, p_amount, CURDATE(), p_mode);
END $$

DELIMITER ;


-- =====================================================
-- PART 6: CREATING TRIGGERS
-- =====================================================

-- TABLE: PAYMENT LOG
-- Purpose: Maintain an audit record of inserted payments.

CREATE TABLE payment_log (
    log_id INT AUTO_INCREMENT PRIMARY KEY, -- Unique log ID
    member_id INT,                         -- Member who made the payment
    amount DECIMAL(10,2),                  -- Amount recorded in the log
    log_date TIMESTAMP DEFAULT CURRENT_TIMESTAMP -- Log creation time
);


-- TRIGGER 1: PAYMENT AUDIT LOG
-- Purpose: Automatically record each new payment in payment_log.
-- NEW refers to the values of the row being inserted.

DELIMITER $$

CREATE TRIGGER trg_payment_log
AFTER INSERT ON payments
FOR EACH ROW
BEGIN
    -- Copy the member ID and payment amount into the audit table.
    INSERT INTO payment_log (member_id, amount)
    VALUES (NEW.member_id, NEW.amount);
END $$

DELIMITER ;


-- TRIGGER 2: VALIDATE PAYMENT AMOUNT
-- Purpose: Prevent payments that are zero or negative.
-- SIGNAL stops the insert and returns an error message.

DELIMITER $$

CREATE TRIGGER trg_validate_payment
BEFORE INSERT ON payments
FOR EACH ROW
BEGIN
    -- Check whether the payment amount is valid.
    IF NEW.amount <= 0 THEN
        SIGNAL SQLSTATE '45000'
        SET MESSAGE_TEXT = 'Payment amount must be greater than zero';
    END IF;
END $$

DELIMITER ;

-- Purpose: Find memberships ending between today and 30 days

FROM memberships
WHERE end_date BETWEEN CURDATE()
                   AND DATE_ADD(CURDATE(), INTERVAL 30 DAY);
